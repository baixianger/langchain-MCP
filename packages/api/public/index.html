QUICK START
Step 1: Install CLI
$ npm i -g langchain-mcp
$ langchain-mcp login
Step 2: Configure Client
$ claude mcp add langchain-mcp -- npx langchain-mcp
{
  "mcpServers": {
    "langchain-mcp": {
      "command": "npx",
      "args": ["langchain-mcp"]
    }
  }
}

{
  "mcpServers": {

    }
  }
}

Config: ~/.cursor/mcp.json

Optional: Check Usage
$ langchain-mcp status
